=== grub.cfg (BIOS)
set shrimp_lang=en
set shrimp_fw="Computer settings"
set shrimp_img=0
# kiwi generated one time grub2 config file
search --file --set=root /boot/0x813228b3
set default=0
if [ -n "$extra_cmdline" ]; then
  submenu "Bootable snapshot $snapshot_num" {
    menuentry "If OK, run snapper rollback and reboot." { true; }
  }
fi
if [ -n "${iso_path}" ]; then
    isoboot="iso-scan/filename=${iso_path}"
fi
set timeout=5
set timeout_style=menu
if [ "${grub_platform}" = "efi" ]; then
    echo "Please press 't' to show the boot menu on this console"
fi
set gfxmode=1920x1080,auto
set font=($root)/boot/x86_64/loader/grub2/fonts/unicode.pf2
set ascii_font=grub2/themes/shrimp/ascii.pf2
set sans_bold_14_font=grub2/themes/shrimp/DejaVuSans-Bold14.pf2
set sans_10_font=grub2/themes/shrimp/DejaVuSans10.pf2
set sans_12_font=grub2/themes/shrimp/DejaVuSans12.pf2
if [ -f ${font} ];then
    loadfont ${font}
fi
if [ -f ($root)/boot/${ascii_font} ];then
    loadfont ($root)/boot/${ascii_font}
fi
if [ -f ($root)/boot/${sans_bold_14_font} ];then
    loadfont ($root)/boot/${sans_bold_14_font}
fi
if [ -f ($root)/boot/${sans_10_font} ];then
    loadfont ($root)/boot/${sans_10_font}
fi
if [ -f ($root)/boot/${sans_12_font} ];then
    loadfont ($root)/boot/${sans_12_font}
fi
if [ -f ($root)/boot/grub2/themes/shrimp/theme.txt ];then
if [ -f ($root)/boot/grub2/shrimp-lang.cfg ]; then
    source ($root)/boot/grub2/shrimp-lang.cfg
fi
if [ -f ($root)/boot/grub2/themes/shrimp/lang/${shrimp_lang}.cfg ]; then
    source ($root)/boot/grub2/themes/shrimp/lang/${shrimp_lang}.cfg
else
    set shrimp_lang=en
fi
if [ "${lockdown}" != "y" -a "${shrimp_img}" != "1" ]; then
        loadfont ($root)/boot/grub2/themes/shrimp/inter-regular-27.pf2
        loadfont ($root)/boot/grub2/themes/shrimp/inter-regular-38.pf2
        loadfont ($root)/boot/grub2/themes/shrimp/inter-semibold-38.pf2
    set theme=($root)/boot/grub2/themes/shrimp/theme-${shrimp_lang}.txt
else
    set theme=($root)/boot/grub2/themes/shrimp/theme-sb-${shrimp_lang}.txt
fi
fi
terminal_input console
terminal_output gfxterm
menuentry "Shrimp" --class os --unrestricted {
    set gfxpayload=keep
    echo Loading kernel...
    linux ($root)/boot/x86_64/loader/linux ${extra_cmdline} ${isoboot} splash quiet loglevel=3 rd.udev.log_level=3 vt.global_cursor_default=0  root=live:CDLABEL=SHRIMP_LIVE rd.live.image rd.live.overlay.persistent rd.live.overlay.cowfs=ext4
    echo Loading initrd...
    initrd ($root)/boot/x86_64/loader/initrd
}
if [ "${grub_platform}" = "efi" ]; then
    menuentry "${shrimp_fw}" --class efi-${shrimp_lang} --unrestricted {
        fwsetup
    }
fi
if [ -f "/.snapshots/grub-snapshot.cfg" ]; then
    source "/.snapshots/grub-snapshot.cfg"
fi
if [ "${grub_platform}" = "efi" ]; then
    hiddenentry "Text mode" --hotkey "t" {
        set textmode=true
        terminal_output console
    }
fi
